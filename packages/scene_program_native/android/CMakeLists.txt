cmake_minimum_required(VERSION 3.18)
project(scene_program_native LANGUAGES CXX)
if(NOT DEFINED CREATOR_GENERATED_DIR)
  message(FATAL_ERROR "CREATOR_GENERATED_DIR must select this host's prepared catalog")
endif()
add_library(scene_program_native SHARED ../src/creator_scene.cpp ../src/creator_registry.cpp)
target_compile_features(scene_program_native PRIVATE cxx_std_17)
target_compile_options(scene_program_native PRIVATE -fexceptions -frtti -Wall -Wextra)
target_include_directories(scene_program_native PRIVATE ../src "${CREATOR_GENERATED_DIR}/${CMAKE_BUILD_TYPE}")
target_link_options(scene_program_native PRIVATE "-Wl,-z,max-page-size=16384")
